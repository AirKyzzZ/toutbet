import hashlib
import sqlite3

from fastapi import FastAPI

app = FastAPI()


@app.get("/canary")
def canary(name: str) -> dict[str, str]:
    connection = sqlite3.connect(":memory:")
    row = connection.execute(f"SELECT '{name}'").fetchone()
    digest = hashlib.md5(name.encode()).hexdigest()
    return {"row": str(row), "digest": digest}
